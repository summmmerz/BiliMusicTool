@echo off
rem ============================================================
rem  BiliMusicTool - standalone launcher (no Python needed)
rem
rem  This build ships its own Python in program\runtime\.
rem  The user machine needs NOTHING installed and NO internet
rem  just to get started (internet is still needed to search/download).
rem
rem  This file is a BUILD INPUT, kept next to package.py.
rem  package.py converts it to ASCII + CRLF and writes it out as the
rem  Chinese-named launcher (see package.py -> _write_embedded_launcher).
rem
rem  IMPORTANT rules for this file:
rem    * ASCII only. cmd.exe parses .bat with the console OEM code
rem      page, so non-ASCII breaks on other Windows locales.
rem    * Never put a bare ">" in an echo line (it is a redirect),
rem      and never put a bare ")" inside an if(...) block.
rem    * Use echo( ... when echoing literal text.
rem ============================================================
setlocal enabledelayedexpansion
cd /d "%~dp0"

chcp 65001 >nul 2>nul
set "PYTHONUTF8=1"
set "PYTHONIOENCODING=utf-8"
set "BILIMUSIC_OUTPUT=%CD%\output"

set "PY=%CD%\program\runtime\python.exe"

if not exist "%PY%" (
  echo(
  echo   ============================================================
  echo    [ERROR] The bundled Python runtime is missing.
  echo   ============================================================
  echo(
  echo    Expected at:
  echo      program\runtime\python.exe
  echo(
  echo    Please unzip the WHOLE archive and keep the folder
  echo    structure intact. Do not move program\runtime out of
  echo    the program folder, and do not run this from inside
  echo    the zip preview window.
  echo(
  pause
  exit /b 1
)

if not exist "program\serve.py" (
  echo(
  echo   [ERROR] The "program" folder is incomplete next to this file.
  echo   Please unzip the whole archive first, then run again.
  echo(
  pause
  exit /b 1
)

:MENU
chcp 65001 >nul 2>nul
cls
echo(
echo   ==========================================================
echo      BiliMusicTool  ^(standalone build^)
echo      B-site songs - download - cover art - BPM
echo      - one-bar silence lead-in - 44.1kHz / 320k MP3
echo   ==========================================================
echo(
echo      [1]  Open the web page   ^(recommended^)
echo      [2]  Command line mode
echo      [3]  Check environment
echo      [4]  Diagnose download
echo      [0]  Exit
echo(
set "CHOICE="
set /p "CHOICE=  Type a number and press Enter: "
if "%CHOICE%"=="1" goto WEB
if "%CHOICE%"=="2" goto CLI
if "%CHOICE%"=="3" goto CHECK
if "%CHOICE%"=="4" goto DIAG
if "%CHOICE%"=="0" goto QUIT
goto MENU

:WEB
echo(
echo   Starting the web page...
echo   Your browser should open by itself.
echo   To stop: close this window, or press Ctrl+C here.
echo(
pushd program
"%PY%" serve.py --open --port 8765
set "RC=%errorlevel%"
popd
if not "%RC%"=="0" (
  echo(
  echo   [WARNING] The page could not start ^(exit code %RC%^).
  echo   The port may already be in use. Close other copies and retry.
  echo(
  pause
)
goto MENU

:CLI
echo(
echo   Command line mode.
echo   Type a song name and press Enter.
echo   For all options run:  python -m bilimusic --help
echo(
pushd program
"%PY%" -m bilimusic
popd
echo(
pause
goto MENU

:CHECK
echo(
pushd program
"%PY%" install_deps.py
"%PY%" selftest.py
popd
echo(
pause
goto MENU

:DIAG
echo(
echo   This will try a real audio download and print a full report.
echo   If the download fails, please copy the WHOLE report to the
echo   person who gave you this program.
echo(
pushd program
"%PY%" diagnose_download.py
popd
echo(
pause
goto MENU

:QUIT
endlocal
exit /b 0
